\usepackage{graphicx}

% Markdown emphasis uses bold instead of italics.
\renewcommand{\emph}[1]{\textbf{#1}}

\pretitle{\begin{center}\rmfamily\CJKfamily{\CJKsfdefault}\bfseries\fontsize{18pt}{25pt}\selectfont}

\preauthor{\begin{center}\normalsize}
\postauthor{\par\end{center}}
\predate{\begin{center}\small}
\postdate{\par\end{center}\vskip 1em}

\usepackage{enumitem}
